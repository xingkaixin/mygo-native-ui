package store

import (
	"sort"
	"strings"
)

// Tags splits a list of tags at commas, trimmed and in lower case, each
// once and sorted.
func Tags(list string) []string {
	seen := map[string]bool{}
	var tags []string
	for _, t := range strings.Split(list, ",") {
		t = strings.ToLower(strings.TrimSpace(t))
		if t == "" || seen[t] {
			continue
		}
		seen[t] = true
		tags = append(tags, t)
	}
	sort.Strings(tags)
	return tags
}

// Tagged returns the notes of s tagged tag.
func (s *Store) Tagged(tag string) []Note {
	s.mu.Lock()
	defer s.mu.Unlock()
	var found []Note
	for _, n := range s.notes {
		for _, t := range n.Tags {
			if t == tag {
				found = append(found, n)
				break
			}
		}
	}
	return found
}
